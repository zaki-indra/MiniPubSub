cmake_minimum_required(VERSION 3.20)
project(MiniPubSub VERSION 1.0.0 LANGUAGES C CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
option(MPS_BUILD_TESTS "Build MiniPubSub tests" ON)
option(MPS_BUILD_SHARED "Build the shared library" ON)
option(MPS_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
option(MPS_BUILD_FUZZERS "Build libFuzzer protocol targets (Clang only)" OFF)
option(MPS_ENABLE_ADDRESS_SANITIZER "Enable AddressSanitizer" OFF)
option(MPS_ENABLE_UNDEFINED_SANITIZER "Enable UndefinedBehaviorSanitizer" OFF)
option(MPS_ENABLE_THREAD_SANITIZER "Enable ThreadSanitizer" OFF)
if(MPS_ENABLE_THREAD_SANITIZER AND (MPS_ENABLE_ADDRESS_SANITIZER OR MPS_ENABLE_UNDEFINED_SANITIZER))
  message(FATAL_ERROR "ThreadSanitizer must be enabled by itself")
endif()
set(MPS_SANITIZERS "")
if(MPS_ENABLE_ADDRESS_SANITIZER)
  list(APPEND MPS_SANITIZERS address)
endif()
if(MPS_ENABLE_UNDEFINED_SANITIZER)
  list(APPEND MPS_SANITIZERS undefined)
endif()
if(MPS_ENABLE_THREAD_SANITIZER)
  list(APPEND MPS_SANITIZERS thread)
endif()
if(MPS_SANITIZERS)
  list(JOIN MPS_SANITIZERS "," MPS_SANITIZER_FLAGS)
  add_compile_options("-fsanitize=${MPS_SANITIZER_FLAGS}" -fno-omit-frame-pointer)
  add_link_options("-fsanitize=${MPS_SANITIZER_FLAGS}")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(MPS_SOURCES src/protocol/codec.cpp src/core/broker.cpp
  src/server/server_runtime.cpp src/client/client_engine.cpp src/c_api/api.cpp)
add_library(minipubsub_obj OBJECT ${MPS_SOURCES})
set_target_properties(minipubsub_obj PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(minipubsub_obj PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(minipubsub_obj PRIVATE MPS_BUILD_SHARED)
target_compile_options(minipubsub_obj PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion $<$<BOOL:${MPS_WARNINGS_AS_ERRORS}>:-Werror>)

add_library(minipubsub_static STATIC $<TARGET_OBJECTS:minipubsub_obj>)
add_library(MiniPubSub::static ALIAS minipubsub_static)
set_target_properties(minipubsub_static PROPERTIES OUTPUT_NAME minipubsub EXPORT_NAME static)
target_include_directories(minipubsub_static PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(minipubsub_static PUBLIC Threads::Threads)
if(MPS_BUILD_SHARED)
  add_library(minipubsub_shared SHARED $<TARGET_OBJECTS:minipubsub_obj>)
  add_library(MiniPubSub::shared ALIAS minipubsub_shared)
  set_target_properties(minipubsub_shared PROPERTIES OUTPUT_NAME minipubsub EXPORT_NAME shared VERSION ${PROJECT_VERSION} SOVERSION 1 LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/cmake/symbol_exports.map)
  target_include_directories(minipubsub_shared PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_options(minipubsub_shared PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/cmake/symbol_exports.map")
  target_link_libraries(minipubsub_shared PUBLIC Threads::Threads)
endif()

if(MPS_BUILD_FUZZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "MPS_BUILD_FUZZERS requires Clang with libFuzzer")
  endif()
  foreach(fuzzer frame_decoder payload_decoder)
    add_executable(mps_${fuzzer}_fuzz fuzz/${fuzzer}_fuzz.cpp)
    target_include_directories(mps_${fuzzer}_fuzz PRIVATE src)
    target_link_libraries(mps_${fuzzer}_fuzz PRIVATE minipubsub_static)
    target_compile_options(mps_${fuzzer}_fuzz PRIVATE -fsanitize=fuzzer)
    target_link_options(mps_${fuzzer}_fuzz PRIVATE -fsanitize=fuzzer)
  endforeach()
endif()

add_executable(minipubsub-server src/cli/server_main.cpp)
target_link_libraries(minipubsub-server PRIVATE minipubsub_static)
add_executable(minipubsub-cli src/cli/client_main.cpp src/cli/client_command.cpp)
target_include_directories(minipubsub-cli PRIVATE src)
target_link_libraries(minipubsub-cli PRIVATE minipubsub_static)
foreach(target minipubsub-server minipubsub-cli)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${MPS_WARNINGS_AS_ERRORS}>:-Werror>)
endforeach()

install(DIRECTORY include/minipubsub DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS minipubsub_static minipubsub-server minipubsub-cli EXPORT MiniPubSubTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(MPS_BUILD_SHARED)
  install(TARGETS minipubsub_shared EXPORT MiniPubSubTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(EXPORT MiniPubSubTargets NAMESPACE MiniPubSub:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MiniPubSub)
configure_package_config_file(cmake/MiniPubSubConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MiniPubSubConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MiniPubSub)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MiniPubSubConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
configure_file(cmake/minipubsub.pc.in minipubsub.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MiniPubSubConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MiniPubSubConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MiniPubSub)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/minipubsub.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)

if(MPS_BUILD_TESTS)
  enable_testing()
  add_executable(mps_protocol_tests tests/unit/protocol_tests.cpp)
  target_include_directories(mps_protocol_tests PRIVATE src)
  target_link_libraries(mps_protocol_tests PRIVATE minipubsub_static)
  add_test(NAME protocol COMMAND mps_protocol_tests)
  add_executable(mps_broker_tests tests/unit/broker_tests.cpp)
  target_include_directories(mps_broker_tests PRIVATE src)
  target_link_libraries(mps_broker_tests PRIVATE minipubsub_static)
  add_test(NAME broker COMMAND mps_broker_tests)
  add_executable(mps_cli_command_tests tests/unit/cli_command_tests.cpp
    src/cli/client_command.cpp)
  target_include_directories(mps_cli_command_tests PRIVATE src)
  target_link_libraries(mps_cli_command_tests PRIVATE minipubsub_static)
  add_test(NAME cli_command COMMAND mps_cli_command_tests)
  add_executable(mps_integration_tests tests/integration/integration_tests.cpp)
  target_link_libraries(mps_integration_tests PRIVATE minipubsub_static)
  add_test(NAME integration COMMAND mps_integration_tests)
  add_executable(mps_cli_integration_tests tests/integration/cli_tests.cpp)
  target_link_libraries(mps_cli_integration_tests PRIVATE minipubsub_static)
  add_dependencies(mps_cli_integration_tests minipubsub-cli)
  add_test(NAME cli_integration COMMAND mps_cli_integration_tests
    $<TARGET_FILE:minipubsub-cli>)
  add_executable(mps_tcp_protocol_tests tests/integration/tcp_protocol_tests.cpp)
  target_include_directories(mps_tcp_protocol_tests PRIVATE src)
  target_link_libraries(mps_tcp_protocol_tests PRIVATE minipubsub_static)
  add_test(NAME tcp_protocol COMMAND mps_tcp_protocol_tests)
  add_executable(mps_c_consumer tests/abi/c_consumer.c)
  if(MPS_BUILD_SHARED)
    target_link_libraries(mps_c_consumer PRIVATE minipubsub_shared)
  else()
    target_link_libraries(mps_c_consumer PRIVATE minipubsub_static)
  endif()
  set_target_properties(mps_c_consumer PROPERTIES LINKER_LANGUAGE CXX)
  target_compile_options(mps_c_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME c_consumer COMMAND mps_c_consumer)
  add_executable(mps_cpp_consumer tests/abi/cpp_consumer.cpp)
  target_link_libraries(mps_cpp_consumer PRIVATE minipubsub_static)
  target_compile_options(mps_cpp_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME cpp_consumer COMMAND mps_cpp_consumer)
  if(MPS_BUILD_SHARED)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
      add_test(NAME python_ctypes_poll COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/abi/python_ctypes_poll.py
        $<TARGET_FILE:minipubsub_shared>)
      add_test(NAME python_ctypes_callback COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/abi/python_ctypes_callback.py
        $<TARGET_FILE:minipubsub_shared>)
    endif()
  endif()
endif()
